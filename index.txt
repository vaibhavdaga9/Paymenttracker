    function escapeHtml(str) {
      if (str === null || str === undefined) return '';
      return String(str)
        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
    }

    function computeNextCallDate(invoiceDateStr, days) {
      var d = new Date(invoiceDateStr);
      if (isNaN(d.getTime())) return '-';
      d.setDate(d.getDate() + (Number(days) || 30));
      var y = d.getFullYear();
      var m = ('0' + (d.getMonth() + 1)).slice(-2);
      var day = ('0' + d.getDate()).slice(-2);
      return y + '-' + m + '-' + day;
    }

    function applyRecordsFilter() {
      var fromVal = document.getElementById('rec_from_date').value;
      var toVal = document.getElementById('rec_to_date').value;
      var searchVal = (document.getElementById('rec_search').value || '').trim().toLowerCase();

      var fromDate = fromVal ? new Date(fromVal) : null;
      var toDate = toVal ? new Date(toVal) : null;
      if (toDate) toDate.setHours(23, 59, 59, 999);

      var filtered = allRecordsCache.filter(function(p) {
        if (searchVal && String(p.clientName || '').toLowerCase().indexOf(searchVal) === -1) return false;

      displayRecords(filtered);

    function updateManager(rowIndex, managerName) {
      showLoading();
      google.script.run
        .withSuccessHandler(function(res) {
          hideLoading();
          if (res.success) {
            // Find the client name for this row from cache
            var clientName = '';
            for (var i = 0; i < allRecordsCache.length; i++) {
              if (allRecordsCache[i].rowIndex === rowIndex) {
                clientName = allRecordsCache[i].clientName || '';
                break;
              }
            }
            var clientKey = clientName.toUpperCase();
            // Propagate to all rows of the same client
            for (var j = 0; j < allRecordsCache.length; j++) {
              var nm = (allRecordsCache[j].clientName || '').toUpperCase();
              if (nm === clientKey) {
                allRecordsCache[j].manager = managerName || '';
              }
            }
            // Re-render so all dropdowns of that client reflect the change
            applyRecordsFilter();
            if (res.updated && res.updated > 1) {
              // Non-blocking notification
              console.log('Manager updated for ' + res.updated + ' invoices of this client.');
            }
          } else {
            alert('Error: ' + (res.message || 'Unable to update manager'));
          }
        })
        .withFailureHandler(function(err) {
          hideLoading();
          alert('Error: ' + err.message);
        })
        .updateManagerForRow(rowIndex, managerName);
    }

    function markReceived(rowIndex) {
      if (!confirm('Mark this payment as Received? This will PERMANENTLY DELETE the record from the sheet.')) return;
      showLoading();
      google.script.run
        .withSuccessHandler(function(res) {
          hideLoading();
          if (res.success) {
            var removedIdx = -1;
            for (var i = 0; i < allRecordsCache.length; i++) {
              if (allRecordsCache[i].rowIndex === rowIndex) { removedIdx = i; break; }
            }
            if (removedIdx !== -1) {
              allRecordsCache.splice(removedIdx, 1);
              for (var j = 0; j < allRecordsCache.length; j++) {
                if (allRecordsCache[j].rowIndex > rowIndex) {
                  allRecordsCache[j].rowIndex -= 1;
                }
              }
            }
            delete selectedRecordRows[rowIndex];
            scheduleGroupsCache = [];
            applyRecordsFilter();
          } else {
            alert('Error: ' + (res.message || 'Unable to mark as received'));
          }
        })
        .withFailureHandler(function(err) {
          hideLoading();
          alert('Error: ' + err.message);
        })
        .markPaymentReceived(rowIndex);
    }

    function applyScheduleFilter() {
      var fromVal = document.getElementById('sched_from_date').value;
      var toVal = document.getElementById('sched_to_date').value;
      var fromDate = fromVal ? new Date(fromVal) : null;
      var toDate = toVal ? new Date(toVal) : null;
      if (toDate) toDate.setHours(23, 59, 59, 999);

      var filtered = scheduleGroupsCache.filter(function(g) {
        if (!fromDate && !toDate) return true;
        var d = new Date(g.date);
        if (isNaN(d.getTime())) return false;
        if (fromDate && d < fromDate) return false;
        if (toDate && d > toDate) return false;
        return true;
      });

      if (currentOverdueFilter !== 'all') {
        var minDaysOverdue = Number(currentOverdueFilter);
        filtered = filtered.filter(function(g) {
          return g.daysOverdue >= minDaysOverdue;
        });
      }

      displaySchedule(filtered);
    }

      document.getElementById('sched_from_date').value = '';
      document.getElementById('sched_to_date').value = '';
      applyScheduleFilter();
    }

    function displaySchedule(groups) {
      var container = document.getElementById('schedule_table_container');
      var totalAll = 0, overdue = 0, filtered = 0;
      
      groups.forEach(function(g) {
        totalAll += g.totalAmount;
        if (g.daysFromToday < 0) overdue += g.totalAmount;
      });

      if (currentOverdueFilter === 'all') {
        filtered = totalAll;
      } else {
        var minDaysOverdue = Number(currentOverdueFilter);
        groups.forEach(function(g) {
          if (g.daysOverdue >= minDaysOverdue) {
            filtered += g.totalAmount;
          }
        });
      }

      document.getElementById('sched_stat_total').textContent = formatCurrency(totalAll);
      document.getElementById('sched_stat_overdue').textContent = formatCurrency(overdue);
      document.getElementById('sched_stat_filtered').textContent = formatCurrency(filtered);

      if (!groups || groups.length === 0) {
        container.innerHTML = '<div class="empty-state">No pending receivables matching your filter 🎉</div>';
        return;
      }

      var html = '<table class="customer-list-table"><thead><tr>' +
        '<th>Invoice Date</th><th># Customers</th><th>Total Amount</th><th>Status</th><th>Call Date</th><th>Days Overdue</th><th>Customers</th>' +
        '</tr></thead><tbody>';

      groups.forEach(function(g) {
        var badge, badgeBg, badgeColor;
        if (g.daysFromToday < 0) {
          badge = Math.abs(g.daysFromToday) + ' day(s) overdue';
          badgeBg = '#fee2e2'; badgeColor = '#991b1b';
        } else if (g.daysFromToday === 0) {
          badge = 'Due today';
          badgeBg = '#fef3c7'; badgeColor = '#92400e';
        } else {
          badge = 'In ' + g.daysFromToday + ' day(s)';
          badgeBg = '#dbeafe'; badgeColor = '#1e40af';
        }
        var statusBadge = '<span style="background:' + badgeBg + '; color:' + badgeColor + '; padding:3px 10px; border-radius:10px; font-size:11px; font-weight:600;">' + badge + '</span>';

        var overdueBadgeClass = '';
        if (g.daysOverdue >= 30) overdueBadgeClass = 'overdue-30d';
        else if (g.daysOverdue >= 21) overdueBadgeClass = 'overdue-21d';
        else if (g.daysOverdue >= 15) overdueBadgeClass = 'overdue-15d';
        else if (g.daysOverdue >= 7) overdueBadgeClass = 'overdue-7d';

        var overdueBadgeHtml = g.daysOverdue > 0
          ? '<span class="overdue-badge ' + overdueBadgeClass + '">+' + g.daysOverdue + ' days</span>'
          : '<span style="color:#6b7280;">—</span>';

        var customersHtml = g.customers.map(function(c) {
          var invLabel = c.remarks ? '<span style="color:#6b7280; font-size:11px;">' + escapeHtml(String(c.remarks).replace(/^Invoice:\s*/i, 'Inv ')) + '</span>' : '';
          return '<div style="font-size:12px; padding:3px 0; border-bottom:1px dashed #e5e7eb;">' +
            '<strong>' + escapeHtml(c.clientName) + '</strong> — ' + formatCurrency(c.amount) +
            (c.mobile ? ' <span style="color:#6b7280;">📱 ' + escapeHtml(c.mobile) + '</span>' : '') +
            (invLabel ? '<br>' + invLabel : '') +
            '</div>';
        }).join('');

        html += '<tr>' +
          '<td><strong>' + g.invoiceDate + '</strong></td>' +
          '<td style="text-align:center;">' + g.count + '</td>' +
          '<td><strong>' + formatCurrency(g.totalAmount) + '</strong></td>' +
          '<td>' + statusBadge + '</td>' +
          '<td><strong>' + g.date + '</strong></td>' +
          '<td>' + overdueBadgeHtml + '</td>' +
          '<td style="min-width:280px;">' + customersHtml + '</td>' +
          '</tr>';
      });

      html += '</tbody></table>';
      container.innerHTML = html;
    }

function searchWhatsAppClient() {
  var searchVal = document.getElementById('wa_search_client').value.trim();
  if (searchVal.length === 0) {
    document.getElementById('wa_results_section').style.display = 'none';
    document.getElementById('wa_empty_state').innerHTML = '<div class="empty-state">Search for a client to get started</div>';
    document.getElementById('wa_empty_state').style.display = 'block';
    return;
  }

  if (!allRecordsCache || allRecordsCache.length === 0) {
    loadWhatsAppRecords();
    return;
  }

  var searchLower = searchVal.toLowerCase();
  var exact = [];
  var partial = [];

  allRecordsCache.forEach(function(payment) {
    if (payment.status && payment.status.toLowerCase() === 'received') return;
    var clientName = String(payment.clientName || '');
    if (!clientName) return;

    if (clientName.toLowerCase() === searchLower) exact.push(payment);
    else if (clientName.toLowerCase().indexOf(searchLower) !== -1) partial.push(payment);
  });

  var matching = exact.length > 0 ? exact : partial;

  if (matching.length === 0) {
    document.getElementById('wa_results_section').style.display = 'none';
    document.getElementById('wa_empty_state').innerHTML = '<div class="empty-state">No pending bills found for "' + escapeHtml(searchVal) + '"</div>';
    document.getElementById('wa_empty_state').style.display = 'block';
    return;
  }

  var selectedClientName = matching[0].clientName;
  matching = matching.filter(function(payment) {
    return String(payment.clientName || '').toLowerCase() === String(selectedClientName || '').toLowerCase();
  });

  waCurrentClient = { name: selectedClientName, mobile: matching[0].mobile };
  waAllBills = matching;
  waSelectedBills = [];

  displayWhatsAppClient(matching);
}

    function displayWhatsAppClient(bills) {
      document.getElementById('wa_empty_state').style.display = 'none';
      document.getElementById('wa_results_section').style.display = 'block';

      var clientName = bills[0].clientName;
      var clientMobile = bills[0].mobile || 'Not Available';

      document.getElementById('wa_client_name').textContent = clientName;
      document.getElementById('wa_client_mobile').textContent = clientMobile;

      var totalDue = bills.reduce(function(sum, b) { return sum + b.amount; }, 0);
      document.getElementById('wa_total_due').textContent = formatCurrency(totalDue);

      var billsHtml = '';
      bills.forEach(function(bill, idx) {
        var isSoon = false;
        if (bill.nextCallDate) {
          var nextCallDate = new Date(bill.nextCallDate);
          var today = new Date();
          today.setHours(0, 0, 0, 0);
          nextCallDate.setHours(0, 0, 0, 0);
          if (nextCallDate.getTime() <= today.getTime()) {
            isSoon = true;
          }
        }

        var billClass = isSoon ? 'title="This invoice is overdue - will be pre-selected"' : '';
        billsHtml += '<div class="whatsapp-bill-item">' +
          '<input type="checkbox" id="wa_bill_' + idx + '" onchange="updateWhatsAppSelection()" ' + (isSoon ? 'checked' : '') + ' ' + billClass + ' />' +
          '<label for="wa_bill_' + idx + '" style="display:none;"></label>' +
          '<div class="bill-details">' +
            '<div class="invoice-no">' + escapeHtml(bill.remarks || 'Invoice') + '</div>' +
            '<div class="invoice-date">' + bill.date + '</div>' +
            '<div class="amount"><strong>' + formatCurrency(bill.amount) + '</strong></div>' +
          '</div>' +
        '</div>';
      });

      document.getElementById('wa_bills_list').innerHTML = billsHtml;
      updateWhatsAppSelection();
    }
